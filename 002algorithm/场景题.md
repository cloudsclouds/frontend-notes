1. 轮播图
本质 = 窗口 + 横向排列的图片容器 + 位移控制
轮播图通过一个 overflow: hidden 的容器作为视口，内部使用 flex 横向排列图片，通过 transform: translateX 移动实现切换，并结合 transition 实现动画效果。
为了实现无缝轮播，会在末尾克隆第一张图片，当滑动到克隆节点时，通过取消动画并重置位置，从而实现连续切换。
同时使用 transform 而不是 left，避免触发布局，提高动画性能。

```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8" />
<title>Carousel</title>

<style>
  .wrapper {
    width: 300px;
    height: 200px;
    margin: 50px auto;
    overflow: hidden;
    position: relative;
  }

  .list {
    display: flex;
    transition: transform 0.5s ease;
  }

  .list img {
    width: 300px;
    height: 200px;
    object-fit: cover;
  }

  button {
    position: absolute;
    top: 50%;
    transform: translateY(-50%);
    padding: 5px 10px;
    cursor: pointer;
  }
</style>
</head>

<body>

<div class="wrapper">
  <div class="list" id="list">
    <img src="https://picsum.photos/id/1015/300/200">
    <img src="https://picsum.photos/id/1016/300/200">
    <img src="https://picsum.photos/id/1018/300/200">
    <!-- 克隆第一张 -->
    <img src="https://picsum.photos/id/1015/300/200">
  </div>

  <button class="prev" onclick="prev()">‹</button>
  <button class="next" onclick="next()">›</button>
</div>

<script>
const list = document.getElementById("list");
const width = 300;
const length = 3; // 原始图片数量
let index = 0;

function next() {
  index++;
  move();
}

function prev() {
  if (index === 0) {
    // 跳到最后一张（无动画）
    list.style.transition = "none";
    list.style.transform = `translateX(-${length * width}px)`;
    index = length;
    
    // 强制重绘（关键！）
    list.offsetHeight;
  }

  index--;
  move();
}

// 移动函数
function move() {
  list.style.transition = "transform 0.5s ease";
  list.style.transform = `translateX(-${index * width}px)`;
}

// 监听动画结束（推荐方式🔥）
list.addEventListener("transitionend", () => {
  // 到达“假的第一张”
  if (index === length) {
    list.style.transition = "none";
    list.style.transform = `translateX(0px)`;
    index = 0;
  }
});
</script>

</body>
</html>
```

2. 用 Vue 写一个倒计时组件
倒计时本质上是一个定时器（setInterval），在组件挂载时启动，定时更新当前剩余时间，倒计时结束后清理定时器。
```vue
<script setup>
import { ref, onMounted, onUnmounted } from 'vue';
let count = ref(10);
let timer = null;

function countDown() {
  timer = setInterval(() => {
    if (count.value > 0) {
      count.value--;
    } else {
      clearInterval(timer)
    }
  }, 1000)
}

onMounted(() => {
  countDown();
})
onUnmounted(() => {
  if (timer) {
    clearInterval(timer);
  }
})
</script>

<template>
  {{ count }}
</template>

```

- 双向绑定

```vue
<template>
  <div>
    <input type="text" v-model="newItem" @keyup.enter="addItem">
    <ul>
      <li v-for="(item, index) in list" :key="index">
        {{ item }} <button @click="removeItem(index)">删除</button>
      </li>
    </ul>
  </div>

<script setup>
import {ref} from 'vue'
const list = ref([])
const newItem = ref('')

  if (newItem.value) {
    list.value.push(newItem.value);
    newItem.value = ''
  }
}
function removeItem(index) {
  list.value.splice(index, 1);
}
</script>
```

6. 表单校验
```js
const rules = {
  username: [
    { required: true, message: '用户名必填' },
    { min: 3, message: '至少3位' }
  ],
  password: [
    { required: true, message: '密码必填' },
    { pattern: /^[0-9a-zA-Z]{6,}$/, message: '至少6位字母或数字' }
  ]
}

function validate(data, rules) {
  const errors = {}

  for (let field in rules) {
    const value = data[field]
    const fieldRules = rules[field]

    for (let rule of fieldRules) {
      // required
      if (rule.required && !value) {
        errors[field] = rule.message
        break
      }

      // min length
      if (rule.min && value.length < rule.min) {
        errors[field] = rule.message
        break
      }

      // pattern
      if (rule.pattern && !rule.pattern.test(value)) {
        errors[field] = rule.message
        break
      }
    }
  }

  return errors
}

function handleSubmit() {
  const errors = validate(formData, rules)

  if (Object.keys(errors).length > 0) {
    console.log('校验失败', errors)
    return
  }

  console.log('提交成功')
}
```

8. 上传图片
```js
// 模拟上传接口：返回 Promise(url)
function upload(file) {
  const formData = new FormData()
  formData.append('file', file)

  return fetch('/upload', {
    method: 'POST',
    body: formData
  })
    .then(res => res.json())
    .then(data => data.url)
}

// 批量上传
async function uploadImages(files) {
  try {
    // 1. 每个文件 → 一个 Promise
    const tasks = Array.from(files).map(file => upload(file))

    // 2. 等全部完成
    const urls = await Promise.all(tasks)

    // 3. 一次性渲染
    render(urls)
  } catch (err) {
    console.error('上传失败：', err)
  }
}

// 渲染函数
function render(urls) {
  const container = document.getElementById('img-container')
  container.innerHTML = urls
    .map(url => `<img src="${url}" />`)
    .join('')
}
```

12. 红绿灯组件
某个路口的红绿灯，会按照红灯亮5s，黄灯亮2s，绿灯亮3s这样的顺序无限循环。要求：每一秒打印当前在亮的灯。
```js
async function light(color, time) {
  for (let i = 0; i < time; i++) {
    console.log(color);
    await new Promise(resolve => setTimeout(resolve, 1000));
  }
}

async function start() {
  while (true) {
    await light('红灯', 5);
    await light('黄灯', 2);
    await light('绿灯', 3);
  }
}

start();
```

12. 自定义定时器 useInterval

封装一个可自动清理的 `setInterval`，避免组件卸载后仍然执行回调。

```jsx
import { useEffect } from 'react';

function useInterval(callback, delay) {
  useEffect(()=> {
    const timer = setInterval(()=> { 
      callback()
    }, delay);

    return () => {
      clearInterval(timer);
    }

  }, [callback, delay])
}
``` 

使用方式：
```jsx
useInterval(() => {
  console.log('每隔一段时间执行一次');
}, 1000);
```
